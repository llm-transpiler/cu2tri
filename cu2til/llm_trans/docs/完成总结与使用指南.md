## 六、快速构建新测试用例

### 方法 1: 手动创建（推荐用于少量算子）

```bash
# 1. 创建目录
CASE="triton_tutorial/my_kernel"
mkdir -p cu2til/cases/$CASE/{triton_,torch_,cute_,logs}

# 2. 实现 Triton kernel (triton_/kernel.py)
cat > cu2til/cases/$CASE/triton_/kernel.py << 'EOF'
import torch
import triton
import triton.language as tl

@triton.jit
def _kernel_impl(...):
    # Your Triton kernel
    pass

def triton_kernel(inputs...):
    # Wrapper
    return output
EOF

# 3. PyTorch 参考 (torch_/ref.py)
cat > cu2til/cases/$CASE/torch_/ref.py << 'EOF'
import torch

def torch_kernel(inputs...):
    # PyTorch reference
    return output
EOF

# 4. 测试数据 (get_data.py)  
cat > cu2til/cases/$CASE/get_data.py << 'EOF'
import torch

class Params:
    def __init__(self):
        self.size = 1024

def get_triton_torch_inputs(params):
    return {'x': torch.randn(..., device='cuda')}

triton_output_tensor_transform = lambda x: x
get_cuda_torch_inputs = get_triton_torch_inputs
cuda_output_tensor_transform = triton_output_tensor_transform
EOF

# 5. 复制测试脚本
cp cu2til/tools/check_cute.py cu2til/cases/$CASE/

# 6. 验证 Triton kernel
cd cu2til/cases/$CASE
conda run -n serve python -c "
import sys; sys.path.insert(0, '.')
from get_data import *
from triton_.kernel import triton_kernel
from torch_.ref import torch_kernel
import torch
inputs = get_triton_torch_inputs(Params())
print('Triton:', triton_kernel(**inputs).shape)
print('Torch:', torch_kernel(**inputs).shape)
print('Diff:', torch.abs(triton_kernel(**inputs) - torch_kernel(**inputs)).max().item())
"
```

## 十五、关键文件模板

### get_data.py 模板

```python
import torch

class Params:
    def __init__(self, **kwargs):
        # Default configuration
        self.size = kwargs.get('size', 1024)
        self.dtype = eval(kwargs.get('dtype', 'torch.float32')) if isinstance(kwargs.get('dtype'), str) else kwargs.get('dtype', torch.float32)

# Multiple test configurations
TEST_CONFIGS = [
    {'size': 1024, 'dtype': 'torch.float32'},
    {'size': 4096, 'dtype': 'torch.float16'},
]

def get_triton_torch_inputs(params: Params):
    torch.manual_seed(0)
    # Generate inputs based on your algorithm
    inputs = {
        'x': torch.randn(params.size, dtype=params.dtype, device='cuda'),
    }
    return inputs

def triton_output_tensor_transform(output):
    return output

get_cuda_torch_inputs = get_triton_torch_inputs
cuda_output_tensor_transform = triton_output_tensor_transform
```
